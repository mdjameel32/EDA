def iqr_outliers(series):
    q1, q3 = series.quantile(0.25), series.quantile(0.75)
    iqr = q3 - q1
    lower, upper = q1 - 1.5 * iqr, q3 + 1.5 * iqr
    return (series < lower) | (series > upper), lower, upper

outlier_table = []
for col in ['Loan Amount (INR)', 'EMI (INR)']:
    mask, low, high = iqr_outliers(df[col])
    df['Outlier - ' + col] = mask
    outlier_table.append({'Column': col, 'Lower fence': round(low), 'Upper fence': round(high),
                          'Outlier rows': mask.sum(), '% of loans': round(mask.mean() * 100, 1)})

outlier_table = pd.DataFrame(outlier_table)

def default_table(col):
    t = df.groupby(col, observed=True)["Is Default"].agg(
        ["count", "sum", "mean"]
    )
    t.columns = ["Loans", "Defaults", "Default Rate %"]
    t["Default Rate %"] = (t["Default Rate %"] * 100).round(1)
    return t.sort_values("Default Rate %", ascending=False)

from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import cross_val_score, StratifiedKFold


X = df[['Credit Score', 'EMI to Income']].copy()
X['EMI to Income'] = X['EMI to Income'].clip(upper=1.5)
X[['Credit Score', 'EMI to Income']] = StandardScaler().fit_transform(X)       # put both on the same scale
X = pd.concat([X, pd.get_dummies(df['Loan Type'], prefix='Type', drop_first=True).astype(int)], axis=1)
y = df['Is Default']

model = LogisticRegression(class_weight='balanced', max_iter=1000)


cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
auc = cross_val_score(model, X, y, cv=cv, scoring='roc_auc')
print(f'Cross-validated ROC-AUC: {auc.mean():.2f} (+/- {auc.std():.2f})   [0.5 = random, 1.0 = perfect]')

model.fit(X, y)
coef = pd.Series(model.coef_[0], index=X.columns).sort_values()

plt.figure(figsize=(8, 5.5))
plt.barh(coef.index, coef.values, color=['RED' if v > 0 else 'TEAL' for v in coef.values])
plt.axvline(0, color='black', linewidth=1)
plt.title('What drives default? (logistic regression coefficients)')
plt.xlabel('Coefficient  (right = raises default risk,  left = lowers it)')
plt.tight_layout()
plt.show()
coef.sort_values(ascending=False).round(2)
